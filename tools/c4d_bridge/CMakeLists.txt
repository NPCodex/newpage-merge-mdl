cmake_minimum_required(VERSION 3.20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(NewPageMergeMDLC4D LANGUAGES C CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Cineware 22 bridge requires the Windows x64 MSVC ABI")
endif()
if(NOT ASSIMP_CINEWARE_ROOT)
  message(FATAL_ERROR "Set ASSIMP_CINEWARE_ROOT to the separately licensed Cineware SDK")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ASSIMP_TOOLS OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(ASSIMP_INSTALL OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_IMPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_ALL_EXPORTERS_BY_DEFAULT OFF CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_GLTF_EXPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_BUILD_NONFREE_C4D_IMPORTER ON CACHE BOOL "" FORCE)
set(ASSIMP_CINEWARE_LIBRARY_POSTFIX "_2019_MT" CACHE STRING "Compatible Cineware 22 static runtime")
set(ASSIMP_BUILD_ZLIB ON CACHE BOOL "" FORCE)
set(ASSIMP_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
add_subdirectory(../../libs/assimp assimp)
add_executable(NewPageMergeMDL-C4D main.cpp)
target_link_libraries(NewPageMergeMDL-C4D PRIVATE assimp)
option(NEWPAGE_C4D_BUILD_FIXTURE "Build the self-authored validation fixture generator" OFF)
if(NEWPAGE_C4D_BUILD_FIXTURE)
  add_executable(c4d-fixture fixture.cpp)
  target_include_directories(c4d-fixture PRIVATE "${ASSIMP_CINEWARE_ROOT}/includes")
  target_link_libraries(c4d-fixture PRIVATE
    "${ASSIMP_CINEWARE_ROOT}/libraries/win/cinewarelib_2019_MT/cinewarelib_release.lib"
    "${ASSIMP_CINEWARE_ROOT}/libraries/win/jpeglib_2019_MT/jpeglib_release.lib"
    ws2_32 winmm)
endif()
