cmake_minimum_required(VERSION 3.20)
project(NewPageOutputTransactionTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
# The test-only pause/throw hooks are compiled here, never in distributed tools.
add_executable(persistent_output_harness ../persistent_output_harness.cpp)
add_executable(ag2_transaction_harness ../ag2_transaction_harness.cpp)
add_executable(ag2_generate_only_harness ../ag2_generate_only_harness.cpp)
add_executable(subtool_transaction_probe ../subtool_transaction_test.cpp)
add_executable(output_transaction_harness ../output_transaction_harness.cpp)
add_executable(ag2_rig_selection_test ../ag2_rig_selection_test.cpp)
target_include_directories(ag2_generate_only_harness PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../libs")
target_include_directories(persistent_output_harness PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../libs")
if(MSVC)
  foreach(target persistent_output_harness ag2_transaction_harness ag2_generate_only_harness subtool_transaction_probe output_transaction_harness)
    target_compile_options(${target} PRIVATE /utf-8)
  endforeach()
endif()
if(MINGW)
  target_link_options(persistent_output_harness PRIVATE -static -municode)
  target_link_options(ag2_transaction_harness PRIVATE -static)
  target_link_options(ag2_generate_only_harness PRIVATE -static)
  target_link_options(subtool_transaction_probe PRIVATE -static -municode)
  target_link_options(output_transaction_harness PRIVATE -static)
endif()
find_package(Threads REQUIRED)
target_link_libraries(persistent_output_harness PRIVATE Threads::Threads)
target_link_libraries(output_transaction_harness PRIVATE Threads::Threads)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME persistent_output COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../persistent_output_test.py" $<TARGET_FILE:persistent_output_harness>)
add_test(NAME ag2_transaction COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../ag2_transaction_test.py" $<TARGET_FILE:ag2_transaction_harness>)
add_test(NAME ag2_generate_only COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../ag2_generate_only_test.py" $<TARGET_FILE:ag2_generate_only_harness>)
add_test(NAME subtool_transaction COMMAND "${Python3_EXECUTABLE}" -c "import pathlib,subprocess,sys,tempfile; work=tempfile.TemporaryDirectory(prefix='newpage-subtool-probe-'); subprocess.run([sys.argv[1],str(pathlib.Path(work.name)/'checks')],check=True)" $<TARGET_FILE:subtool_transaction_probe>)
add_test(NAME subtool_durable COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../subtool_durable_test.py" $<TARGET_FILE:subtool_transaction_probe>)
add_test(NAME compiler_output_transaction COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../output_transaction_test.py" $<TARGET_FILE:output_transaction_harness>)
add_test(NAME ag2_rig_selection COMMAND $<TARGET_FILE:ag2_rig_selection_test>)
set_tests_properties(persistent_output ag2_transaction ag2_generate_only subtool_transaction compiler_output_transaction ag2_rig_selection PROPERTIES TIMEOUT 180)
set_tests_properties(subtool_durable PROPERTIES TIMEOUT 180)
